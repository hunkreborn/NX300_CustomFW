sha256sum <ELFs, DB e scripts selecionados>
arm-linux-gnueabi-readelf -h -n -d -r -s -Ws <ELF>
arm-linux-gnueabi-nm -n -C <ELF>
arm-linux-gnueabi-objdump -d -C <ELF>
arm-linux-gnueabi-objdump -d -C --start-address=<addr> --stop-address=<addr> <ELF>
arm-linux-gnueabi-addr2line -a -f -C -e di-camera-app-nx300-full <addresses>
arm-linux-gnueabi-readelf --debug-dump=decodedline di-camera-app-nx300-full
strings -a -t x <ELFs selecionados>
rg -a -n -i 'autoexec|start_script|stop_script|/mnt/mmc|/sdcard|launchpad|di-camera-app' <corpus selecionado>
sed -n <ranges> <scripts/configs>
python3 (sqlite3 em URI mode=ro para ler .app_info.db)
cmp e sha256sum para correlacionar variantes
